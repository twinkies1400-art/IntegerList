#ifndef INTEGERLIST_H
#define INTEGERLIST_H

#include <vector>

class IntegerList {
private:
    std::vector<int> data;

public:
    // Member A
    void add(int value);
    int getSize() const;

    // Member B
    void display() const;
    void clear();

    // Member C
    bool removeLast();
    int find(int value) const;
};

#endif

#include "IntegerList.h"
#include <iostream>

// Member A implementations below

// Member B implementations below

// Member C implementations below

void IntegerList::add(int value) {
    data.push_back(value);
}

int IntegerList::getSize() const {
    return static_cast<int>(data.size());
}

void IntegerList::display() const {
    std::cout << "[ ";
    for (int val : data) {
        std::cout << val << " ";
    }
    std::cout << "]" << std::endl;
}

void IntegerList::clear() {
    data.clear();
}
bool IntegerList::removeLast() {
    if (data.empty()) {
        return false;
    }
    data.pop_back();
    return true;
}

int IntegerList::find(int value) const {
    for (size_t i = 0; i < data.size(); ++i) {
        if (data[i] == value) {
            return static_cast<int>(i);
        }
    }
    return -1;
}
